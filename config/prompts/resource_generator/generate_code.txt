UDP不可靠传输与应用层重传实验
演示UDP丢包场景下的应用层可靠性保证策略
参考：RFC 768 - User Datagram Protocol
\"\"\"
import socket
import random
import time
import threading

MAX_RETRIES = 3        # 最大重传次数
TIMEOUT = 1.0          # 重传超时(秒)
SIMULATE_LOSS = 0.3    # 模拟30%丢包率

def unreliable_server(host='127.0.0.1', port=8888):
    \"\"\"模拟不可靠的UDP服务端（随机丢弃请求模拟网络丢包）\"\"\"
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.bind((host, port))
    print(f'[Server] UDP服务端启动于 {host}:{port}')
    
    while True:
        data, addr = sock.recvfrom(1024)
        # 模拟网络丢包：30%概率不回复
        if random.random() < SIMULATE_LOSS:
            print(f'[Server] 模拟丢包！丢弃来自{addr}的请求')
            continue
        seq = data.decode().split(':')[0]
        print(f'[Server] 收到seq={seq}，发送ACK')
        sock.sendto(f'ACK:{seq}'.encode(), addr)

def reliable_client(host='127.0.0.1', port=8888):
    \"\"\"带重传机制的UDP客户端（应用层实现可靠传输）\"\"\"
    time.sleep(0.3)
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.settimeout(TIMEOUT)
    
    for seq in range(1, 6):
        message = f'{seq}:Hello UDP #{seq}'
        delivered = False
        for attempt in range(1, MAX_RETRIES + 1):
            sock.sendto(message.encode(), (host, port))
            print(f'[Client] 发送 seq={seq} (尝试 {attempt}/{MAX_RETRIES})')
            try:
                ack, _ = sock.recvfrom(1024)
                if f'ACK:{seq}' in ack.decode():
                    print(f'[Client] 收到ACK seq={seq} ✓')
                    delivered = True
                    break
            except socket.timeout:
                print(f'[Client] 超时！{TIMEOUT}s内未收到ACK，准备重传...')
        if not delivered:
            print(f'[Client] seq={seq} 发送失败（{MAX_RETRIES}次重传均超时）')
    sock.close()

if __name__ == '__main__':
    print('=== UDP应用层重传实验（模拟30%丢包）===')
    t = threading.Thread(target=unreliable_server, daemon=True)
    t.start()
    reliable_client()",
